npm install express
Express is used to create the REST API. Step 6: Create the API server
Server.js
const express = require("express"); const app = express();
const PORT = 3000;
app.get("/add", (req, res) => {
const a = Number(req.query.a); const b = Number(req.query.b); res.json({
operation: "addition",
result: a + b });
});
app.get("/multiply", (req, res) => {
const a = Number(req.query.a); const b = Number(req.query.b); res.json({
operation: "multiplication",
result: a * b });
});
app.listen(PORT, () => {
console.log(`Calculator API running at http://localhost:${PORT}`); });
Step 8: Test the Addition API
Open another terminal and run: curl "http://localhost:3000/add?a=10&b=5" Step 9: Test the Multiplication API
Run: curl http://localhost:3000/multiply?a=10&b=5
