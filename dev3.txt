/server.js
const express = require('express');
const cors = require('cors');
const path = require('path');

const app = express();
const PORT = 3000;

app.use(cors());
app.use(express.json());
app.use(express.static(__dirname));

// Direct fallback to open your HTML form smoothly
app.get('/', (req, res) => {
    res.sendFile(path.join(__dirname, 'Exp1reg.html')); 
});

// Captures form entries over the local network bridge
app.post('/api/register', (req, res) => {
    const userData = req.body;

    console.log(`\n [Microservice Dashboard] New Registration Saved!`);
    console.log(`Name: ${userData.firstName} ${userData.lastName}`);
    console.log(`Email: ${userData.email}`);

    res.json({
        status: "Success",
        message: `Thank you, ${userData.firstName}! Your registration was successful.`
    });
});

app.listen(PORT, () => {
    console.log(`\nServer running smoothly with zero external client script files!`);
    console.log(` Go to your browser link: http://localhost:${PORT}\n`);
});
